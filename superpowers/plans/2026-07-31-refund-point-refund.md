# Refund-Point Refund Creation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let a refund-point user in `super-app` pay out a traveller's refundable tags from their phone — the flow web already offers at `operations/refund`.

**Architecture:** One hidden route `(auth)/refund` renders a single scrolling screen: traveller → refund method + payment details → refundable tag multi-select → optional signature pads, with a pinned bar showing the running refund total and a confirm sheet before submit. All state lives in one hook (`useRefundFlow`); all arithmetic and payload construction live in a pure, unit-tested module (`refund.logic.ts`). Three API calls go through `src/actions/**` wrappers, one of which requires generating a ContractService client the app does not yet have.

**Tech Stack:** React Native 0.81 + Expo 54, expo-router (file-based, typed routes), NativeWind/Tailwind, TypeScript strict, Jest (`jest-expo`), `@gorhom/bottom-sheet`, generated OpenAPI clients under `src/saas/`.

**Spec:** `docs/superpowers/specs/2026-07-31-refund-point-refund-design.md`

```ts
   * Present for refund-point staff accounts. Same repeated-claim caveat as
```

```ts
import useUserStore from "@/store/user";
import { useMemo } from "react";

/**
 * Refund point the signed-in user is currently acting for, or `null` when the
 * account has no refund-point claim.
 *
 * Mirrors `useMerchantId`: the claim is a string or an array depending on how
 * many refund points the account is affiliated with, and switching affiliation
 * issues a new token, so reading it from the store keeps this in sync.
 */
export function useRefundPointId(): string | null {
  const { user } = useUserStore();
  return useMemo(() => {
    const refundPointId = user?.jwtUser?.RefundPointId;
    if (!refundPointId) return null;
    return Array.isArray(refundPointId) ? refundPointId[0] || null : refundPointId;
  }, [user]);
}
```

```ts
/**
```

```ts
import type { UniRefund_TagService_Tags_TagListItemDto } from "@/saas/TagService";
import { isExpiredCard, luhnValid, onlyDigits, parseExpiry } from "@/utils/card/card";

/** Card as typed into the form: display-formatted number, "MM/YY" expiry. */
export type CardEntry = { number: string; expiry: string };

/** Captured signatures, already converted to base64 at submit time. */
export type SignaturesBase64 = { traveller?: string; refundPoint?: string };

export type RefundFlowState = {
  refundPointId: string | null;
  selectedTags: UniRefund_TagService_Tags_TagListItemDto[];
  method: RefundMethod | null;
  card: CardEntry;
  /** File URIs while the form is open; converted to base64 only at submit. */
  signatures: { traveller?: string; refundPoint?: string };

export type RefundTotals = {
  salesAmount: number;
  vatAmount: number;
  refundFee: number;

/**
 * Ledger for the selected rows.
 *
 * The currency comes from the first row, matching web. The UI stops a
 * mixed-currency selection from being made in the first place, because summing
 * across currencies produces a number that means nothing at a cash desk.
 */
export function sumTagTotals(
  rows: UniRefund_TagService_Tags_TagListItemDto[],
): RefundTotals {
  const add = (key: "salesAmount" | "vatAmount" | "refundFee" | "refund") =>
    rows.reduce((sum, row) => sum + (row[key] ?? 0), 0);

  return {
    salesAmount: add("salesAmount"),
    vatAmount: add("vatAmount"),
    refundFee: add("refundFee"),
    refund: add("refund"),
    currency: rows[0]?.currency || "USD",
  };
}

/** Luhn plus a not-yet-expired "MM/YY". `now` is injectable for the tests. */
export function isCardEntryValid(card: CardEntry, now: Date = new Date()): boolean {
  if (!luhnValid(card.number)) return false;
  const parsed = parseExpiry(card.expiry);
  if (!parsed) return false;
  return !isExpiredCard(parsed.month, parsed.year, now);
}

/** Card details are only required by the method that actually carries them. */
function methodNeedsCard(method: RefundMethod | null): boolean {
  return method === "CreditCard";
}

export function canSubmit(state: RefundFlowState, now: Date = new Date()): boolean {
  if (!state.refundPointId) return false;
  if (state.selectedTags.length === 0) return false;
  if (!state.method) return false;
  if (state.isSubmitting) return false;
  if (methodNeedsCard(state.method) && !isCardEntryValid(state.card, now)) return false;
  return true;
}

/**
 * The create payload, or null when the state is not submittable.
 *
 * Only the block belonging to the chosen method is sent: `paidDate` for cash,
 * `paidCardDetail` for a card payout already taken on the terminal, neither for
 * the remaining contracted methods — which is exactly what web sends.
 */
export function buildCreateRefundDto(
  state: RefundFlowState,
  signaturesBase64: SignaturesBase64,
  paidDate: string,
): UniRefund_RefundService_Refunds_CreateRefundDto | null {
  if (!canSubmit(state) || !state.refundPointId || !state.method) return null;

  const dto: UniRefund_RefundService_Refunds_CreateRefundDto = {
    refundPointId: state.refundPointId,
    tagIds: state.selectedTags.map((row) => row.id),
    refundTypeEnum: state.method,
  };

  if (state.method === "Cash") {
    dto.paidDate = paidDate;
  }

  if (state.method === "CreditCard") {
    const parsed = parseExpiry(state.card.expiry);
    if (!parsed) return null;
    dto.paidCardDetail = {
      maskedNumber: onlyDigits(state.card.number),
      cardExpiryMonth: parsed.month,
      cardExpiryYear: parsed.year,
    };
  }

  if (signaturesBase64.traveller) {
    dto.travellerSignatureBase64 = signaturesBase64.traveller;
  }
  if (signaturesBase64.refundPoint) {
    dto.refundPointSignatureBase64 = signaturesBase64.refundPoint;
  }

  return dto;
}
```

Note: `canSubmit` is called inside `buildCreateRefundDto` without a `now`, so it validates against the real clock at submit — correct, since a card that expired between form-fill and submit should not go through.

```ts
import { getPaymentTypesByRefundPointIdApi } from "@/actions/ContractService/actions";
import { postRefundApi } from "@/actions/RefundService/post";
import { getRefundableTagsApi } from "@/actions/TagService/actions";
import { useRefundPointId } from "@/hooks/useRefundPointId";
import type { UniRefund_TagService_Tags_TagListItemDto } from "@/saas/TagService";
import type { UniRefund_TagService_Travellers_TravellerRequestDto } from "@/saas/TagService";
import type { SignatureTarget } from "@/screens/shared/_components/tag-calculator/line";
import { logger } from "@/utils/logger";
import { File } from "expo-file-system";
import { useCallback, useEffect, useMemo, useState } from "react";
import {
  buildCreateRefundDto,
  canSubmit as canSubmitState,
  normalizeCardExpiry,
  sumTagTotals,
  type CardEntry,
  type RefundMethod,
} from "./refund.logic";

type Traveller = UniRefund_TagService_Travellers_TravellerRequestDto;
type TagRow = UniRefund_TagService_Tags_TagListItemDto;

async function toBase64(uri?: string): Promise<string | undefined> {
  if (!uri) return undefined;
  return await new File(uri).base64();
}

/**
 * Owns one in-progress refund: the traveller, the method and its payment
 * details, the tag selection, signatures, and the create call.
 *
 * Component state rather than a store, for the same reason `useCreateTag`
 * gives: the traveller picker is an inline modal, so nothing has to survive a
 * navigation.
 */
export function useRefundFlow() {
  const refundPointId = useRefundPointId();

  const [methods, setMethods] = useState<RefundMethod[] | null>(null);
  const [methodsError, setMethodsError] = useState(false);
  const [traveller, setTravellerState] = useState<Traveller | null>(null);
  const [method, setMethodState] = useState<RefundMethod | null>(null);
  const [card, setCard] = useState<CardEntry>({ number: "", expiry: "" });
  const [isExportValidated, setIsExportValidatedState] = useState(true);
  const [tags, setTags] = useState<TagRow[]>([]);
  const [tagsLoading, setTagsLoading] = useState(false);
  const [tagsError, setTagsError] = useState(false);
  const [selectedIds, setSelectedIds] = useState<string[]>([]);
  const [signatures, setSignatures] = useState<
    Partial<Record<SignatureTarget, string>>
  >({});
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [createdRefund, setCreatedRefund] = useState<{
    tagCount: number;
    method: RefundMethod;
    amount: number;
    currency: string;
  } | null>(null);

  const loadMethods = useCallback(async () => {
    if (!refundPointId) return;
    setMethodsError(false);
    try {
      const result = await getPaymentTypesByRefundPointIdApi(refundPointId);
      const list = result ?? [];
      setMethods(list);
      // A single contracted method is not a choice — preselect it, as web does.
      if (list.length === 1) setMethodState(list[0] ?? null);
    } catch (error) {
      logger.warn("[Refund] failed to load payment types", error);
      setMethods(null);
      setMethodsError(true);
    }
  }, [refundPointId]);

  // Network fetch keyed on an external value (the refund point claim). No
  // derived state or user event to hang it off — an effect is the honest tool.
  useEffect(() => {
    void loadMethods();
  }, [loadMethods]);

  const documentNumber = traveller?.travellerDocumentNumber;

  const loadTags = useCallback(async () => {
    if (!refundPointId || !documentNumber || !method) {
      setTags([]);
      return;
    }
    setTagsLoading(true);
    setTagsError(false);
    try {
      const result = await getRefundableTagsApi({
        refundPointId,
        travellerDocumentNumber: documentNumber,
        refundType: method,
        isExportValidated,
        maxResultCount: 100,
      });
      setTags(result?.items ?? []);
    } catch (error) {
      logger.warn("[Refund] failed to load refundable tags", error);
      setTags([]);
      setTagsError(true);
    } finally {
      setTagsLoading(false);
    }
  }, [refundPointId, documentNumber, method, isExportValidated]);

  // Same justification as `loadMethods`: keyed on the query, not on a render.
  useEffect(() => {
    void loadTags();
  }, [loadTags]);

  /**
   * Any change to the tag query invalidates the selection — the list about to
   * arrive may not contain what was ticked. Cleared at the point of change
   * rather than in an effect, so it can never lag a render behind the list.
   */
  const setMethod = useCallback((next: RefundMethod) => {
    setMethodState(next);
    setSelectedIds([]);
  }, []);

  const setIsExportValidated = useCallback((next: boolean) => {
    setIsExportValidatedState(next);
    setSelectedIds([]);
  }, []);

  /** A new traveller invalidates the tags and the signatures taken for the old one. */
  const setTraveller = useCallback((next: Traveller | null) => {
    setTravellerState(next);
    setSelectedIds([]);
    setSignatures({});
  }, []);

  const setCardNumber = useCallback((next: string) => {
    setCard((prev) => ({ ...prev, number: next }));
  }, []);

  const setCardExpiry = useCallback((next: string) => {
    setCard((prev) => ({ ...prev, expiry: normalizeCardExpiry(next) }));
  }, []);

  const selectedTags = useMemo(
    () => tags.filter((row) => selectedIds.includes(row.id)),
    [tags, selectedIds],
  );

  const totals = useMemo(() => sumTagTotals(selectedTags), [selectedTags]);

  /**
   * Once something is selected, only tags in the same currency may join it.
   * Summing across currencies produces a total that means nothing at the desk.
   */
  const isTagSelectable = useCallback(
    (row: TagRow) => {
      const anchor = selectedTags[0]?.currency;
      if (!anchor) return true;
      return (row.currency || "") === anchor;
    },
    [selectedTags],
  );

  const toggleTag = useCallback(
    (row: TagRow) => {
      setSelectedIds((prev) =>
        prev.includes(row.id)
          ? prev.filter((id) => id !== row.id)
          : [...prev, row.id],
      );
    },
    [],
  );

  const setSignature = useCallback((target: SignatureTarget, uri: string) => {
    setSignatures((prev) => ({ ...prev, [target]: uri }));
  }, []);

  const flowState = useMemo(
    () => ({
      refundPointId,
      selectedTags,
      method,
      card,
      signatures,
      isSubmitting,
    }),
    [refundPointId, selectedTags, method, card, signatures, isSubmitting],
  );

  const canSubmit = useMemo(() => canSubmitState(flowState), [flowState]);

  /** Resolves true on success. The caller owns the error toast. */
  const submit = useCallback(async (): Promise<boolean> => {
    if (!canSubmitState(flowState) || !method) return false;
    setIsSubmitting(true);
    try {
      const [travellerSignature, refundPointSignature] = await Promise.all([
        toBase64(signatures.traveller),
        toBase64(signatures.refundPoint),
      ]);
      const dto = buildCreateRefundDto(
        flowState,
        {
          traveller: travellerSignature,
          refundPoint: refundPointSignature,
        },
        new Date().toISOString(),
      );
      if (!dto) return false;

      await postRefundApi(dto);
      setCreatedRefund({
        tagCount: selectedTags.length,
        method,
        amount: totals.refund,
        currency: totals.currency,
      });
      return true;
    } catch (error) {
      logger.warn("[Refund] refund creation failed", error);
      return false;
    } finally {
      setIsSubmitting(false);
    }
  }, [flowState, method, signatures, selectedTags, totals]);

  /** Back to an empty form for the next customer, keeping the loaded methods. */
  const reset = useCallback(() => {
    setTravellerState(null);
    setSelectedIds([]);
    setSignatures({});
    setCard({ number: "", expiry: "" });
    setTags([]);
    setCreatedRefund(null);
  }, []);

  return {
    refundPointId,
    methods,
    methodsError,
    reloadMethods: loadMethods,
    traveller,
    setTraveller,
    method,
    setMethod,
    card,
    setCardNumber,
    setCardExpiry,
    isExportValidated,
    setIsExportValidated,
    tags,
    tagsLoading,
    tagsError,
    reloadTags: loadTags,
    selectedIds,
    toggleTag,
    isTagSelectable,
    selectedTags,
    totals,
    signatures,
    setSignature,
    canSubmit,
    isSubmitting,
    submit,
    createdRefund,
    reset,
  };
}
```

```ts
/**
 * Keeps the expiry field in the exact `MM/YY` shape `parseExpiry` requires,
 * inserting the slash as the operator types and refusing anything longer.
 */
export function normalizeCardExpiry(raw: string): string {
  const digits = onlyDigits(raw).slice(0, 4);
  if (digits.length <= 2) return digits;
  return `${digits.slice(0, 2)}/${digits.slice(2)}`;
}
```

```tsx
import { Ionicons, type IoniconsTypes } from "@/components/Ionicons";
import Input from "@/components/Input";
import { useLocalization } from "@/providers/LocalizationProvider";
import type { TranslationKey } from "@/localization/config";
import { formatCardNumber } from "@/utils/card/card";
import { cn } from "@/utils/cn";
import { Pressable, Text, View } from "react-native";
import {
  isCardEntryValid,
  type CardEntry,
  type RefundMethod,
} from "../refund.logic";

const METHOD_ICONS: Record<RefundMethod, IoniconsTypes> = {
  Cash: "cash-outline",
  CreditCard: "card-outline",
  BankTransfer: "business-outline",
  Wallet: "wallet-outline",
  CashViaPartner: "people-outline",
  IbanTransfer: "swap-horizontal-outline",
};

const METHOD_LABELS: Record<RefundMethod, TranslationKey> = {
  Cash: "MobileApp.Refund.MethodCash",
  CreditCard: "MobileApp.Refund.MethodCreditCard",
  BankTransfer: "MobileApp.Refund.MethodBankTransfer",
  Wallet: "MobileApp.Refund.MethodWallet",
  CashViaPartner: "MobileApp.Refund.MethodCashViaPartner",
  IbanTransfer: "MobileApp.Refund.MethodIbanTransfer",
};

/**
 * Method tiles plus the details that method needs.
 *
 * Sits above the tag list because `refundType` filters the refundable-tags
 * query — the method chosen decides which tags are offered.
 */
export function MethodPicker({
  methods,
  methodsError,
  onRetry,
  method,
  onSelect,
  card,
  onCardNumberChange,
  onCardExpiryChange,
  disabled,
}: {
  methods: RefundMethod[] | null;
  methodsError: boolean;
  onRetry: () => void;
  method: RefundMethod | null;
  onSelect: (method: RefundMethod) => void;
  card: CardEntry;
  onCardNumberChange: (value: string) => void;
  onCardExpiryChange: (value: string) => void;
  disabled?: boolean;
}) {
  const { t, formatDate } = useLocalization();

  if (methodsError) {
    return (
      <View className="rounded-2xl border border-gray-200 bg-white p-4">
        <Text className="text-sm text-gray-600 mb-3">
          {t("MobileApp.Refund.MethodsError")}
        </Text>
        <Pressable
          onPress={onRetry}
          accessibilityRole="button"
          className="self-start px-4 py-2 rounded-full bg-gray-100"
        >
          <Text className="text-sm font-medium text-gray-700">
            {t("MobileApp.Refund.Retry")}
          </Text>
        </Pressable>
      </View>
    );
  }

  if (methods && methods.length === 0) {
    return (
      <View className="rounded-2xl border border-gray-200 bg-white p-4">
        <Text className="text-sm text-gray-600">
          {t("MobileApp.Refund.MethodsEmpty")}
        </Text>
      </View>
    );
  }

  const showCardError = method === "CreditCard" && card.number.length > 0 && !isCardEntryValid(card);

  return (
    <View className="rounded-2xl border border-gray-200 bg-white p-4">
      <Text className="text-xs font-medium text-muted mb-2">
        {t("MobileApp.Refund.Method")}
      </Text>

      <View className="flex-row flex-wrap gap-2">
        {(methods ?? []).map((option) => {
          const active = method === option;
          return (
            <Pressable
              key={option}
              onPress={() => onSelect(option)}
              disabled={disabled}
              accessibilityRole="button"
              accessibilityState={{ selected: active }}
              className={cn(
                "flex-1 basis-24 items-center gap-1.5 rounded-xl border p-3",
                active ? "border-primary bg-primary/5" : "border-gray-200",
                disabled && "opacity-50",
              )}
            >
              <Ionicons
                name={METHOD_ICONS[option]}
                size={20}
                color={active ? "#db0000" : "#6B7280"}
              />
              <Text
                className={cn(
                  "text-xs font-medium text-center",
                  active ? "text-primary" : "text-gray-600",
                )}
              >
                {t(METHOD_LABELS[option])}
              </Text>
            </Pressable>
          );
        })}
      </View>

      {method === "Cash" && (
        <View className="mt-3 flex-row items-center justify-between">
          <Text className="text-sm text-muted">
            {t("MobileApp.Refund.PaidDate")}
          </Text>
          {/* Stamped at submit, so this is a statement of fact, not an input. */}
          <Text className="text-sm font-medium text-foreground">
            {formatDate(new Date().toISOString())}
          </Text>
        </View>
      )}

      {method === "CreditCard" && (
        <View className="mt-3">
          <Input
            title={t("MobileApp.Refund.CardNumber")}
            iconName="card-outline"
            value={card.number}
            onChangeText={(value) => onCardNumberChange(formatCardNumber(value))}
            keyboardType="number-pad"
            editable={!disabled}
            placeholder="0000 0000 0000 0000"
          />
          <Input
            title={t("MobileApp.Refund.CardExpiry")}
            iconName="calendar-outline"
            value={card.expiry}
            onChangeText={onCardExpiryChange}
            keyboardType="number-pad"
            editable={!disabled}
            placeholder="MM/YY"
            maxLength={5}
          />
          {showCardError && (
            <Text className="text-xs text-primary">
              {t("MobileApp.Refund.CardInvalid")}
            </Text>
          )}
        </View>
      )}
    </View>
  );
}
```

```tsx
import { Ionicons } from "@/components/Ionicons";
import { useLocalization } from "@/providers/LocalizationProvider";
import type { UniRefund_TagService_Tags_TagListItemDto } from "@/saas/TagService";
import { cn } from "@/utils/cn";
import { ActivityIndicator, Pressable, Text, View } from "react-native";

type TagRow = UniRefund_TagService_Tags_TagListItemDto;

/**
 * The traveller's refundable tags, with the export-validation filter above
 * them. Rows in a currency other than the one already selected are disabled —
 * a mixed-currency total is not a number anyone can hand over.
 */
export function RefundableTagList({
  tags,
  isLoading,
  hasError,
  onRetry,
  isExportValidated,
  onExportValidatedChange,
  selectedIds,
  onToggle,
  isTagSelectable,
  disabled,
}: {
  tags: TagRow[];
  isLoading: boolean;
  hasError: boolean;
  onRetry: () => void;
  isExportValidated: boolean;
  onExportValidatedChange: (value: boolean) => void;
  selectedIds: string[];
  onToggle: (row: TagRow) => void;
  isTagSelectable: (row: TagRow) => boolean;
  disabled?: boolean;
}) {
  const { t, formatCurrency } = useLocalization();

  return (
    <View className="rounded-2xl border border-gray-200 bg-white p-4">
      <Text className="text-xs font-medium text-muted mb-2">
        {t("MobileApp.Refund.Tags")}
      </Text>

      <View className="flex-row gap-2 mb-3">
        {[true, false].map((value) => {
          const active = isExportValidated === value;
          return (
            <Pressable
              key={String(value)}
              onPress={() => onExportValidatedChange(value)}
              disabled={disabled}
              accessibilityRole="button"
              accessibilityState={{ selected: active }}
              className={cn(
                "flex-1 rounded-full border py-2",
                active ? "border-primary bg-primary/5" : "border-gray-200",
              )}
            >
              <Text
                className={cn(
                  "text-xs font-medium text-center",
                  active ? "text-primary" : "text-gray-600",
                )}
              >
                {t(
                  value
                    ? "MobileApp.Refund.StatusExportValidated"
                    : "MobileApp.Refund.StatusNeedsValidation",
                )}
              </Text>
            </Pressable>
          );
        })}
      </View>

      {isLoading && <ActivityIndicator className="my-4" />}

      {!isLoading && hasError && (
        <View>
          <Text className="text-sm text-gray-600 mb-3">
            {t("MobileApp.Refund.TagsError")}
          </Text>
          <Pressable
            onPress={onRetry}
            accessibilityRole="button"
            className="self-start px-4 py-2 rounded-full bg-gray-100"
          >
            <Text className="text-sm font-medium text-gray-700">
              {t("MobileApp.Refund.Retry")}
            </Text>
          </Pressable>
        </View>
      )}

      {!isLoading && !hasError && tags.length === 0 && (
        <Text className="text-sm text-muted py-2">
          {t("MobileApp.Refund.TagsEmpty")}
        </Text>
      )}

      {!isLoading &&
        !hasError &&
        tags.map((row) => {
          const selected = selectedIds.includes(row.id);
          const selectable = selected || isTagSelectable(row);
          return (
            <Pressable
              key={row.id}
              onPress={() => onToggle(row)}
              disabled={disabled || !selectable}
              accessibilityRole="checkbox"
              accessibilityState={{ checked: selected, disabled: !selectable }}
              className={cn(
                "flex-row items-center gap-3 py-3 border-b border-gray-100",
                !selectable && "opacity-40",
              )}
            >
              <Ionicons
                name={selected ? "checkbox" : "square-outline"}
                size={22}
                color={selected ? "#db0000" : "#9CA3AF"}
              />
              <View className="flex-1">
                <Text className="text-sm font-medium text-foreground">
                  {row.tagNumber}
                </Text>
                <Text className="text-xs text-muted">{row.merchantTitle}</Text>
              </View>
              <Text className="text-sm font-semibold text-foreground">
                {formatCurrency(row.refund ?? 0, row.currency || "USD")}
              </Text>
            </Pressable>
          );
        })}

      {!isLoading && !hasError && selectedIds.length > 0 && tags.some((row) => !isTagSelectable(row)) && (
        <Text className="text-xs text-muted mt-2">
          {t("MobileApp.Refund.MixedCurrency")}
        </Text>
      )}
    </View>
  );
}
```

```tsx
  const signatureSheetRef = useRef<BottomSheetModal>(null);

    useState<SignatureTarget | null>(null);

  const openSignature = useCallback((target: SignatureTarget) => {
    setSignatureTarget(target);
    signatureSheetRef.current?.present();
  }, []);

  const handleConfirm = useCallback(async () => {
    const ok = await flow.submit();
    confirmSheetRef.current?.dismiss();
    if (!ok) {
      // The form, the selection and the signatures all survive a failure —
      // nothing was paid out, so nothing the operator entered should be lost.
      toastRef.current?.show("error", t("MobileApp.Refund.SubmitError"));
    }
  }, [flow, t, toastRef]);

  if (!flow.refundPointId) {
    return (
      <TabPage title={t("MobileApp.Refund.Title")}>
        <View className="flex-1 items-center justify-center px-6">
          <Text className="text-muted text-center text-base">
            {t("MobileApp.Refund.NoRefundPoint")}
          </Text>
        </View>
      </TabPage>
    );
  }

  if (flow.createdRefund) {
    return (
      <TabPage title={t("MobileApp.Refund.Title")}>
        <RefundSuccess
          tagCount={flow.createdRefund.tagCount}
          amount={flow.createdRefund.amount}
          currency={flow.createdRefund.currency}
          method={flow.createdRefund.method}
          onNewRefund={flow.reset}
        />
      </TabPage>
    );
  }

  return (
    <TabPage title={t("MobileApp.Refund.Title")}>
      <View className="flex-1">
        <ScrollView
          className="flex-1"
          contentContainerClassName="gap-3 pb-4"
          showsVerticalScrollIndicator={false}
          keyboardShouldPersistTaps="handled"
        >
          <TravellerSearchCard
            traveller={flow.traveller}
            onSelect={flow.setTraveller}
            disabled={flow.isSubmitting}
          />

          <MethodPicker
            methods={flow.methods}
            methodsError={flow.methodsError}
            onRetry={flow.reloadMethods}
            method={flow.method}
            onSelect={flow.setMethod}
            card={flow.card}
            onCardNumberChange={flow.setCardNumber}
            onCardExpiryChange={flow.setCardExpiry}
            disabled={flow.isSubmitting}
          />

          {flow.traveller && flow.method ? (
            <RefundableTagList
              tags={flow.tags}
              isLoading={flow.tagsLoading}
              hasError={flow.tagsError}
              onRetry={flow.reloadTags}
              isExportValidated={flow.isExportValidated}
              onExportValidatedChange={flow.setIsExportValidated}
              selectedIds={flow.selectedIds}
              onToggle={flow.toggleTag}
              isTagSelectable={flow.isTagSelectable}
              disabled={flow.isSubmitting}
            />
          ) : null}

          {flow.selectedIds.length > 0 ? (
            <SignaturePads
              signatures={flow.signatures}
              onSign={openSignature}
              targets={SIGNATURE_TARGETS}
            />
          ) : null}
        </ScrollView>

        <RefundSummaryBar
          selectedCount={flow.selectedIds.length}
          totals={flow.totals}
          canSubmit={flow.canSubmit}
          onPress={() => confirmSheetRef.current?.present()}
        />
      </View>

      <RefundConfirmSheet
        sheetRef={confirmSheetRef}
        selectedCount={flow.selectedIds.length}
        method={flow.method}
        totals={flow.totals}
        isSubmitting={flow.isSubmitting}
        onConfirm={handleConfirm}
        onCancel={() => confirmSheetRef.current?.dismiss()}
      />

      <SignatureSheet
        sheetRef={signatureSheetRef}
        target={signatureTarget}
        onSave={flow.setSignature}
      />
    </TabPage>
  );
}
```
