## Appendix A: `scripts/find-missing-i18n.mjs`

This is the validated script for Task 4. It was run against this repo during planning and produced `NEEDS_KEY 174 / SUPPRESSED 120 / ABP_PLUMBING 23 / unresolved 5`.

```js
#!/usr/bin/env node
/**
 * Reports localization keys that SchemaForm derives from a JSON schema but that do
 * not exist in the service's resource files.
 *
 * createUiSchemaWithResource({ schema, resources, name }) walks the schema and sets
 * each property's `ui:title` from resources[`${name}.${property}`]. A missing key
 * degrades to lodash.startCase(property), so the field renders an English-ish label
 * instead of breaking - invisible in English, wrong in Turkish.
 *
 * Not every derived key is needed. A nested createUiSchemaWithResource spliced in via
 * `extend` overrides the outer pass for that property, and a custom ui:field/ui:widget
 * or displayLabel:false means the title never renders. Both are excluded from NEEDS_KEY.
 *
 * Design: the sibling unirefund/docs repo, at
 * docs/superpowers/specs/2026-08-05-schemaform-grid-i18n-coverage-design.md
 *
 *   node scripts/find-missing-i18n.mjs --app=web
 *   node scripts/find-missing-i18n.mjs --app=web --json
 */
import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";

const SCRIPT_DIR = path.dirname(fileURLToPath(import.meta.url));
const SOURCE_EXTENSIONS = /\.(?:ts|tsx)$/;
const SKIP_DIRECTORIES = new Set(["node_modules", ".next", ".turbo", "dist"]);
const RESOURCE_SCOPES = ["core", "unirefund"];
const MAX_DEPTH = 4;

/** ABP infrastructure properties that should be filtered out of a form, not labelled. */
const ABP_PLUMBING = new Set([
  "extraProperties",
  "concurrencyStamp",
  "entityVersion",
  "isDeleted",
  "deleterId",
  "deletionTime",
  "lastModifierId",
  "creatorId",
]);

function parseArgs(argv) {
  const args = { app: "web", json: false, root: path.resolve(SCRIPT_DIR, "..") };
  for (const arg of argv) {
    if (arg === "--json") args.json = true;
    else if (arg.startsWith("--app=")) args.app = arg.slice("--app=".length);
    else if (arg.startsWith("--root=")) args.root = path.resolve(arg.slice("--root=".length));
    else throw new Error(`Unknown argument: ${arg}`);
  }
  if (!["web", "ssr"].includes(args.app)) {
    throw new Error(`--app must be "web" or "ssr", received "${args.app}"`);
  }
  return args;
}

function collectSourceFiles(dir, files = []) {
  if (!fs.existsSync(dir)) return files;
  for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
    if (SKIP_DIRECTORIES.has(entry.name)) continue;
    const full = path.join(dir, entry.name);
    if (entry.isDirectory()) collectSourceFiles(full, files);
    else if (SOURCE_EXTENSIONS.test(entry.name)) files.push(full);
  }
  return files;
}

/** The `{...}` starting at `start`, respecting nesting. */
function balancedBraces(src, start) {
  let depth = 0;
  for (let i = start; i < src.length; i += 1) {
    if (src[i] === "{") depth += 1;
    else if (src[i] === "}") {
      depth -= 1;
      if (depth === 0) return src.slice(start, i + 1);
    }
  }
  return null;
}

/**
 * Value text of a top-level `key:` inside an object literal, ignoring nested
 * braces. A fixed look-ahead window would read a sibling call's value instead.
 */
function topLevelProp(objectLiteral, key) {
  const body = objectLiteral.slice(1, -1);
  const parts = [];
  let depth = 0;
  let current = "";
  for (const char of body) {
    if ("{[(".includes(char)) depth += 1;
    else if ("}])".includes(char)) depth -= 1;
    if (char === "," && depth === 0) {
      parts.push(current);
      current = "";
      continue;
    }
    current += char;
  }
  parts.push(current);
  const matcher = new RegExp(`^\\s*${key}\\s*:\\s*([\\s\\S]*)$`);
  for (const part of parts) {
    const found = matcher.exec(part);
    if (found) return found[1].trim();
  }
  return null;
}

/** Top-level `name: value` entries of an object literal, as [name, valueText] pairs. */
function topLevelEntries(objectLiteral) {
  const body = objectLiteral.slice(1, -1);
  const entries = [];
  let depth = 0;
  let current = "";
  for (const char of body) {
    if ("{[(".includes(char)) depth += 1;
    else if ("}])".includes(char)) depth -= 1;
    if (char === "," && depth === 0) {
      entries.push(current);
      current = "";
      continue;
    }
    current += char;
  }
  entries.push(current);
  const out = [];
  for (const entry of entries) {
    const found = /^\s*"?([A-Za-z_$][A-Za-z0-9_$]*)"?\s*:\s*([\s\S]*)$/.exec(entry);
    if (found) out.push([found[1], found[2].trim()]);
  }
  return out;
}

/** Every `export const $Foo = {...}` across the repo, generated or hand-written. */
function indexSchemas(root, app) {
  const literals = new Map();
  const files = [
    ...collectSourceFiles(path.join(root, "packages")),
    ...collectSourceFiles(path.join(root, "apps", app, "src")),
  ];
  for (const file of files) {
    const src = fs.readFileSync(file, "utf8");
    if (!src.includes("$")) continue;
    for (const match of src.matchAll(/(?:export\s+)?const (\$[A-Za-z0-9_]+)\s*=\s*/g)) {
      const brace = src.indexOf("{", match.index + match[0].length);
      if (brace === -1 || brace > match.index + match[0].length + 2) continue;
      const literal = balancedBraces(src, brace);
      if (literal && !literals.has(match[1])) literals.set(match[1], literal);
    }
  }
  return literals;
}

function makeSchemaResolver(literals) {
  const cache = new Map();
  return function resolve(id) {
    if (cache.has(id)) return cache.get(id);
    let value = null;
    const literal = literals.get(id);
    if (literal) {
      try {
        value = new Function(`return (${literal.replace(/\s+as\s+const\s*$/, "")});`)();
      } catch {
        value = null;
      }
    }
    cache.set(id, value);
    return value;
  };
}

/**
 * Keys uiSchemaFromSchema would look up, classified. `overridden` and `suppressed`
 * hold top-level property names whose derived titles never reach the screen.
 */
function deriveKeys(schema, prefix, { overridden, suppressed }, depth = 0, out = []) {
  if (!schema || depth > MAX_DEPTH) return out;
  const properties = schema.properties || schema.items?.properties;
  if (!properties) return out;

  for (const [property, value] of Object.entries(properties)) {
    const key = `${prefix}.${property}`;
    const topLevel = depth === 0;
    if (topLevel && overridden.has(property)) continue; // nested call re-derives this subtree

    let kind = "NEEDS_KEY";
    if (ABP_PLUMBING.has(property)) kind = "ABP_PLUMBING";
    else if (topLevel && suppressed.has(property)) kind = "SUPPRESSED";
    out.push({ key, kind });

    if (value && typeof value === "object") {
      const childContext = {
        overridden: new Set(),
        suppressed:
          topLevel && suppressed.has(property) ? new Set(Object.keys(value.properties || {})) : new Set(),
      };
      if (Array.isArray(value.enum)) {
        for (const member of value.enum) {
          if (typeof member === "string") {
            out.push({ key: `${key}.${member}`, kind: kind === "NEEDS_KEY" ? "NEEDS_KEY" : kind });
          }
        }
      }
      if (value.properties || value.items?.properties) {
        const nested = deriveKeys(value, key, childContext, depth + 1, []);
        for (const entry of nested) {
          out.push(kind === "NEEDS_KEY" ? entry : { ...entry, kind });
        }
      }
    }
  }
  return out;
}

function loadServices(root, app) {
  const services = {};
  const base = path.join(root, "apps", app, "src", "language-data");
  for (const scope of RESOURCE_SCOPES) {
    const dir = path.join(base, scope);
    if (!fs.existsSync(dir)) continue;
    for (const service of fs.readdirSync(dir)) {
      const file = path.join(dir, service, "resources/en.json");
      if (fs.existsSync(file)) {
        services[service] = new Set(Object.keys(JSON.parse(fs.readFileSync(file, "utf8"))));
      }
    }
  }
  return services;
}

function analyze(root, app) {
  const resolveSchema = makeSchemaResolver(indexSchemas(root, app));
  const services = loadServices(root, app);
  const findings = [];
  const unresolved = [];

  for (const file of collectSourceFiles(path.join(root, "apps", app, "src"))) {
    if (file.includes(`${path.sep}language-data${path.sep}`)) continue;
    const src = fs.readFileSync(file, "utf8");
    if (!src.includes("createUiSchemaWithResource")) continue;
    const rel = path.relative(root, file).split(path.sep).join("/");

    const aliases = new Map();
    for (const m of src.matchAll(/(\$[A-Za-z0-9_]+)\s+as\s+(\$[A-Za-z0-9_]+)/g)) aliases.set(m[2], m[1]);
    const fromImports = [...src.matchAll(/language-data\/(?:core|unirefund)\/([A-Za-z]+)/g)].map((m) => m[1]);
    const fromT = [...src.matchAll(/\bt\.([A-Za-z]+)\b/g)].map((m) => m[1]);

    for (const call of src.matchAll(/createUiSchemaWithResource\(\s*\{/g)) {
      const brace = src.indexOf("{", call.index);
      const args = balancedBraces(src, brace);
      if (!args) {
        unresolved.push({ file: rel, reason: "unbalanced argument object" });
        continue;
      }

      const nameRaw = topLevelProp(args, "name");
      const name = nameRaw ? (/^"([^"]+)"/.exec(nameRaw)?.[1] ?? null) : "Form";
      if (name === null) {
        unresolved.push({ file: rel, reason: `name is not a string literal: ${nameRaw.slice(0, 40)}` });
        continue;
      }

      const schemaRaw = topLevelProp(args, "schema");
      const expr = schemaRaw ? /^(\$[A-Za-z0-9_]+(?:\.[A-Za-z0-9_]+)*)/.exec(schemaRaw)?.[1] : null;
      if (!expr) {
        unresolved.push({
          file: rel,
          name,
          reason: `schema is not a resolvable $identifier: ${(schemaRaw || "none").slice(0, 40)}`,
        });
        continue;
      }

      const [rootRaw, ...pathParts] = expr.split(".");
      let schema = resolveSchema(aliases.get(rootRaw) ?? rootRaw);
      if (!schema) {
        unresolved.push({
          file: rel,
          name,
          reason: `${aliases.get(rootRaw) ?? rootRaw} not found in the schema index`,
        });
        continue;
      }
      for (const segment of pathParts) schema = schema?.[segment];
      if (!schema) {
        unresolved.push({ file: rel, name, reason: `property path ${expr} did not resolve` });
        continue;
      }

      const resourcesRaw = topLevelProp(args, "resources");
      let service = null;
      const viaT = /^t\.([A-Za-z]+)/.exec(resourcesRaw || "");
      if (viaT && services[viaT[1]]) service = viaT[1];
      if (!service) {
        const candidates = [...new Set([...fromImports, ...fromT])].filter((s) => services[s]);
        if (candidates.length === 1) [service] = candidates;
        else {
          unresolved.push({
            file: rel,
            name,
            reason: `resource service ambiguous (${candidates.join(", ") || "none found"})`,
          });
          continue;
        }
      }

      // extend tells us which properties the outer pass does not own
      const overridden = new Set();
      const suppressed = new Set();
      const extendRaw = topLevelProp(args, "extend");
      if (extendRaw && extendRaw.startsWith("{")) {
        const extend = balancedBraces(extendRaw, 0);
        if (extend) {
          for (const [property, literal] of topLevelEntries(extend)) {
            if (literal.includes("createUiSchemaWithResource")) overridden.add(property);
            else if (/"ui:field"|"ui:widget"|displayLabel:\s*false/.test(literal)) suppressed.add(property);
          }
        }
      }

      for (const entry of deriveKeys(schema, name, { overridden, suppressed })) {
        if (services[service].has(entry.key)) continue;
        findings.push({ file: rel, service, name, key: entry.key, kind: entry.kind });
      }
    }
  }
  return { findings, unresolved };
}

function main() {
  const { app, json, root } = parseArgs(process.argv.slice(2));
  const { findings, unresolved } = analyze(root, app);

  const distinct = new Map();
  for (const f of findings) {
    const id = `${f.service}::${f.key}`;
    // NEEDS_KEY wins over a softer classification from another call site
    if (!distinct.has(id) || f.kind === "NEEDS_KEY") distinct.set(id, f);
  }
  const byKind = { NEEDS_KEY: [], SUPPRESSED: [], ABP_PLUMBING: [] };
  for (const f of distinct.values()) byKind[f.kind].push(f);

  if (json) {
    console.log(
      JSON.stringify(
        {
          needsKey: byKind.NEEDS_KEY,
          suppressed: byKind.SUPPRESSED,
          abpPlumbing: byKind.ABP_PLUMBING,
          unresolved,
        },
        null,
        1
      )
    );
    return byKind.NEEDS_KEY.length === 0 ? 0 : 1;
  }

  console.log(`app=${app}  distinct missing keys=${distinct.size}  unresolved call sites=${unresolved.length}`);
  console.log(`  NEEDS_KEY    ${byKind.NEEDS_KEY.length}  (label renders, no key exists)`);
  console.log(`  SUPPRESSED   ${byKind.SUPPRESSED.length}  (custom widget or label hidden)`);
  console.log(`  ABP_PLUMBING ${byKind.ABP_PLUMBING.length}  (should be filtered out of the form, not translated)`);

  const byService = {};
  for (const f of byKind.NEEDS_KEY) (byService[f.service] ??= []).push(f.key);
  if (byKind.NEEDS_KEY.length) {
    console.log(`\nNEEDS_KEY by service:`);
    for (const [service, keys] of Object.entries(byService).sort((a, b) => b[1].length - a[1].length)) {
      console.log(`\n  ${service} (${keys.length})`);
      for (const key of keys.sort()) console.log(`    ${key}`);
    }
  }
  if (unresolved.length) {
    console.log(`\nUNRESOLVED - these were not checked, so this report is not proof of coverage:`);
    for (const u of unresolved) console.log(`  ${u.file}${u.name ? ` name="${u.name}"` : ""}: ${u.reason}`);
  }
  return byKind.NEEDS_KEY.length === 0 ? 0 : 1;
}

try {
  process.exit(main());
} catch (error) {
  console.error(`find-missing-i18n: ${error.message}`);
  process.exit(2);
}
```

## Appendix B: `scripts/verify-i18n-queue.mjs`

Validated during planning: reports 14 waiting / 0 translated / 0 broken with exit 0 on the real queue, and correctly reports `TRANSLATED but still queued: 1` with exit 1 when a queued key's `tr` is changed.

```js
#!/usr/bin/env node
/**
 * Verifies scripts/i18n-untranslated.json against the resource files.
 *
 * Keys in that queue carry the English string in tr.json as a deliberate
 * placeholder - the project's choice is that a translator fills them, not that
 * Claude invents Turkish. A tr value equal to its en value is indistinguishable
 * from a legitimately identical translation ("Email", "Fax"), so the queue is the
 * only record of which ones are still waiting. This check keeps that record honest:
 *
 *   - a queued key that no longer has tr == en has been TRANSLATED; the entry is
 *     stale and must be removed, which is a loud failure rather than silent rot.
 *   - a queued key missing from either locale file, or naming an unknown service,
 *     is a broken entry.
 *
 * The reverse direction - a key with tr == en that is NOT queued - is deliberately
 * not checked. Hundreds of existing keys are legitimately identical across locales,
 * so that check would be noise rather than signal.
 *
 *   node scripts/verify-i18n-queue.mjs [--app=web] [--root=<path>]
 */
import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";

const SCRIPT_DIR = path.dirname(fileURLToPath(import.meta.url));
const RESOURCE_SCOPES = ["core", "unirefund"];

function parseArgs(argv) {
  const args = { app: "web", root: path.resolve(SCRIPT_DIR, "..") };
  for (const arg of argv) {
    if (arg.startsWith("--app=")) args.app = arg.slice("--app=".length);
    else if (arg.startsWith("--root=")) args.root = path.resolve(arg.slice("--root=".length));
    else throw new Error(`Unknown argument: ${arg}`);
  }
  return args;
}

function resourceDir(root, app, service) {
  for (const scope of RESOURCE_SCOPES) {
    const dir = path.join(root, "apps", app, "src", "language-data", scope, service, "resources");
    if (fs.existsSync(dir)) return dir;
  }
  return null;
}

function main() {
  const { app, root } = parseArgs(process.argv.slice(2));
  const queueFile = path.join(root, "scripts", "i18n-untranslated.json");
  if (!fs.existsSync(queueFile)) {
    console.log("no queue file - nothing to verify");
    return 0;
  }
  const queue = JSON.parse(fs.readFileSync(queueFile, "utf8"));

  const translated = [];
  const broken = [];
  let waiting = 0;

  const cache = new Map();
  for (const entry of queue) {
    const dir = resourceDir(root, app, entry.service);
    if (!dir) { broken.push({ ...entry, why: "unknown service" }); continue; }
    if (!cache.has(dir)) {
      cache.set(dir, {
        en: JSON.parse(fs.readFileSync(path.join(dir, "en.json"), "utf8")),
        tr: JSON.parse(fs.readFileSync(path.join(dir, "tr.json"), "utf8")),
      });
    }
    const { en, tr } = cache.get(dir);
    if (!(entry.key in en)) { broken.push({ ...entry, why: "absent from en.json" }); continue; }
    if (!(entry.key in tr)) { broken.push({ ...entry, why: "absent from tr.json" }); continue; }
    if (tr[entry.key] !== en[entry.key]) {
      translated.push({ ...entry, tr: tr[entry.key] });
      continue;
    }
    waiting += 1;
  }

  console.log(`queue entries: ${queue.length}`);
  console.log(`  still awaiting translation (tr == en): ${waiting}`);
  console.log(`  TRANSLATED but still queued:           ${translated.length}`);
  console.log(`  BROKEN entries:                        ${broken.length}`);

  if (translated.length) {
    console.log(`\nThese have been translated - remove them from the queue:`);
    for (const e of translated) console.log(`  ${e.service} :: ${e.key}  tr=${JSON.stringify(e.tr)}`);
  }
  if (broken.length) {
    console.log(`\nBroken entries:`);
    for (const e of broken) console.log(`  ${e.service} :: ${e.key} - ${e.why}`);
  }
  return translated.length === 0 && broken.length === 0 ? 0 : 1;
}

try {
  process.exit(main());
} catch (error) {
  console.error(`verify-i18n-queue: ${error.message}`);
  process.exit(2);
}
```
